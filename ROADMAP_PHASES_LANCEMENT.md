# 🛠️ IMPLÉMENTATION TECHNIQUE

## Configuration des phases

Créer un fichier de configuration :

```dart
// lib/config/app_phases.dart

enum AppPhase {
  phase1_mvp,
  phase2_premium,
  phase3_social,
  phase4_ai,
  phase5_coach_pro,
}

class AppPhaseConfig {
  // CHANGER CETTE VALEUR À CHAQUE MISE À JOUR
  static const AppPhase currentPhase = AppPhase.phase1_mvp;
  
  static bool isFeatureAvailable(String featureId) {
    switch (currentPhase) {
      case AppPhase.phase1_mvp:
        return _phase1Features.contains(featureId);
      case AppPhase.phase2_premium:
        return _phase1Features.contains(featureId) || 
               _phase2Features.contains(featureId);
      case AppPhase.phase3_social:
        return _phase1Features.contains(featureId) || 
               _phase2Features.contains(featureId) ||
               _phase3Features.contains(featureId);
      case AppPhase.phase4_ai:
        return _phase1Features.contains(featureId) || 
               _phase2Features.contains(featureId) ||
               _phase3Features.contains(featureId) ||
               _phase4Features.contains(featureId);
      case AppPhase.phase5_coach_pro:
        return true; // Tout est disponible
    }
  }
  
  static const _phase1Features = [
    'dashboard', 'food_journal', 'calculator', 'recipes_basic',
    'recipe_explorer', 'exercise_library', 'create_workout_basic',
    'workout_session', 'calendar_basic', 'body_analysis_basic',
    'health_injuries', 'profile', 'settings',
  ];
  
  static const _phase2Features = [
    'meal_planner', 'shopping_list', 'recipes_unlimited',
    'workouts_unlimited', 'history_unlimited', 'stats_advanced',
    'programs_premium', 'export_data', 'coach_personalities',
  ];
  
  static const _phase3Features = [
    'chat_match', 'visio_training', 'community_chat',
    'social_feed', 'follows', 'events', 'sport_gaming',
    'directory', 'map', 'referral', 'memo',
  ];
  
  static const _phase4Features = [
    'food_scan', 'coach_ia', 'alter_ego', 'transformation_ia',
    'ai_suggestions', 'ai_program', 'ai_voice_coach',
  ];
  
  static const _phase5Features = [
    'coach_business', 'client_management', 'client_tracking',
    'coach_messaging', 'coach_planning', 'live_courses',
    'video_packs', 'replays', 'invoicing', 'coach_vs_coach',
    'hard_challenge',
  ];
}
```
